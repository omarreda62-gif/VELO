export type StockStatus = 'available' | 'limited' | 'out_of_stock'; export interface PriceTier { minQty: number; label: string; // مثال: "1 - 5 عبوات" أو "كرتونة (6+)" unitPrice: number; // سعر البيع للعميل merchantCost: number;// تكلفة الشراء من التاجر } export interface Product { id: string; merchantId: string; merchantName: string; // يُستخدم داخلياً في لوحة الإدارة فقط category: 'cafe_dessert' | 'restaurant' | 'pumps_bottles' | 'bundles'; brand: string; nameAr: string; unitLabel: string; unitWeightGrams?: number; // وزن العبوة بالجرام لحساب تكلفة الجرعة defaultDoseGrams?: number;// وزن البمب أو الجرعة التقريبية stockStatus: StockStatus; tiers: PriceTier[]; } export interface CartItem { product: Product; qty: number; } export interface Zone { id: string; nameAr: string; deliveryFee: number; freeDeliveryThreshold: number | null; } // 1. اختيار شريحة السعر المناسبة للكمية export function getActiveTier(product: Product, qty: number): PriceTier { const sorted = [...product.tiers].sort((a, b) => b.minQty - a.minQty); return sorted.find((t) => qty >= t.minQty) || product.tiers[0]; } // 2. تقدير تكلفة الجرعة / البمب على الكوب أو الوجبة export function estimateDoseCost(product: Product, qty: number): number | null { if (!product.unitWeightGrams || !product.defaultDoseGrams) return null; const tier = getActiveTier(product, qty); const totalGrams = product.unitWeightGrams * qty; const dosesCount = totalGrams / product.defaultDoseGrams; if (dosesCount <= 0) return null; return Number(((tier.unitPrice * qty) / dosesCount).toFixed(2)); } // 3. حساب رسوم التوصيل حسب المركز وقيمة الطلب export function calculateDeliveryFee(zone: Zone | null, subtotal: number): number { if (!zone || subtotal <= 0) return 0; if (zone.freeDeliveryThreshold !== null && subtotal >= zone.freeDeliveryThreshold) { return 0; } return zone.deliveryFee; } // 4. تقسيم السلة تلقائياً إلى "أوامر استلام" منفصلة لكل تاجر للإدارة export interface MerchantPickupOrder { merchantId: string; merchantName: string; merchantPayoutTotal: number; items: { productId: string; nameAr: string; brand: string; qty: number; merchantCost: number; lineCost: number; }[]; } export function buildMerchantPickupOrders(cart: CartItem[]): MerchantPickupOrder[] { const grouped = new Map(); for (const item of cart) { const tier = getActiveTier(item.product, item.qty); const lineCost = tier.merchantCost * item.qty; if (!grouped.has(item.product.merchantId)) { grouped.set(item.product.merchantId, { merchantId: item.product.merchantId, merchantName: item.product.merchantName, merchantPayoutTotal: 0, items: [], }); } const pickup = grouped.get(item.product.merchantId)!; pickup.items.push({ productId: item.product.id, nameAr: item.product.nameAr, brand: item.product.brand, qty: item.qty, merchantCost: tier.merchantCost, lineCost, }); pickup.merchantPayoutTotal += lineCost; } return Array.from(grouped.values()); }
